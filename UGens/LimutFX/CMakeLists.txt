cmake_minimum_required(VERSION 3.25)
project(DexterLimutFX LANGUAGES CXX)

if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()

set(SC_SOURCE_DIR "" CACHE PATH "SuperCollider source/header tree")
set(SC_PLUGIN_INTERFACE_DIR "" CACHE PATH
    "Directory containing SC_PlugIn.h (overrides SC_SOURCE_DIR)")
if(SC_PLUGIN_INTERFACE_DIR)
  set(SC_PLUGIN_INCLUDE "${SC_PLUGIN_INTERFACE_DIR}")
else()
  set(SC_PLUGIN_INCLUDE "${SC_SOURCE_DIR}/include/plugin_interface")
endif()
if(NOT EXISTS "${SC_PLUGIN_INCLUDE}/SC_PlugIn.h")
  message(FATAL_ERROR
    "Set SC_SOURCE_DIR or SC_PLUGIN_INTERFACE_DIR to SuperCollider plugin headers.")
endif()

add_library(LimutFX MODULE LimutFX.cpp)
target_compile_features(LimutFX PRIVATE cxx_std_17)
target_include_directories(LimutFX PRIVATE "${SC_PLUGIN_INCLUDE}")
get_filename_component(SC_INCLUDE_ROOT "${SC_PLUGIN_INCLUDE}" DIRECTORY)
if(EXISTS "${SC_INCLUDE_ROOT}/common")
  target_include_directories(LimutFX PRIVATE "${SC_INCLUDE_ROOT}/common")
endif()
if(EXISTS "${SC_SOURCE_DIR}/include/common")
  target_include_directories(LimutFX PRIVATE "${SC_SOURCE_DIR}/include/common")
endif()
if(EXISTS "${SC_SOURCE_DIR}/common")
  target_include_directories(LimutFX PRIVATE "${SC_SOURCE_DIR}/common")
endif()
set_target_properties(LimutFX PROPERTIES PREFIX "")
if(WIN32)
  set_target_properties(LimutFX PROPERTIES SUFFIX ".scx")
endif()

add_executable(limutfx-dsp-tests tests.cpp)
target_compile_features(limutfx-dsp-tests PRIVATE cxx_std_17)
target_include_directories(limutfx-dsp-tests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
enable_testing()
add_test(NAME limutfx-dsp COMMAND limutfx-dsp-tests)
